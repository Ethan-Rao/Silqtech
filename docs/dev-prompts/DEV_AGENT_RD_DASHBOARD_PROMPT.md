## Architecture

### Data Layers

**Layer 1 — Base project data (from Excel)**
`public/data/rd/projects.json` — generated by a parse script from the Excel file. Unlike NuSilq, this file IS auto-regenerated on each update (the Excel is the single source of truth). Do not hand-edit it.

**Layer 2 — File manifest**
`public/data/rd/files.json` — generated by the file-sync script. Lists every downloadable file per project with its filename and description. Stored in git and regenerated when files are synced.

**Layer 3 — Cloud notes overlay (Spaces)**
`raoeqms-files/rd/notes.json` — browser-added notes synced to DigitalOcean Spaces, identical pattern to NuSilq. API routes at `/api/rd/notes`.

**Layer 4 — Project files (Spaces)**
`raoeqms-files/rd-files/[slug]/[filename]` — actual downloadable files uploaded by the user from local `R&DDashboard/[folder]/`. Served via presigned URLs from `/api/rd/download`.

## Part 1 — Scripts

### 1a — `scripts/generate-rd-data.mjs`

Parse `R&DDashboard/Silq 2026 NRE Projects.xlsx` and write `public/data/rd/projects.json`.

```javascript
// scripts/generate-rd-data.mjs
import XLSX from 'xlsx'
import { join, dirname } from 'path'
import { fileURLToPath } from 'url'
import { writeFileSync, mkdirSync } from 'fs'

const __dirname = dirname(fileURLToPath(import.meta.url))
const ROOT = join(__dirname, '..')
const XLSX_PATH = join(ROOT, '..', 'R&DDashboard', 'Silq 2026 NRE Projects.xlsx')
const OUT_PATH = join(ROOT, 'public', 'data', 'rd', 'projects.json')

// Slugify: lowercase, replace spaces/slashes/special chars with hyphens, trim
function slugify(name) {
  // For "Ernesto Andrade/Momentum" → use last segment after /
  const parts = name.split('/')
  const base = parts[parts.length - 1].trim()
  return base.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '')
}

// Manual overrides for slugs that need special handling
const SLUG_OVERRIDES = {
  'Ernesto Andrade/Momentum': 'momentum',
  'P-Trap': 'p-trap',
  'Design 33': 'design-33',
  'Liqid Medical': 'liqid-medical',
  'Mayo Clinic': 'mayo-clinic',
  'Boston Scientific': 'boston-scientific',
}

const wb = XLSX.readFile(XLSX_PATH, { cellDates: false })
const ws = wb.Sheets['Active']
const rows = XLSX.utils.sheet_to_json(ws, { defval: '' })

const tier1 = [], tier2 = [], tier3 = []

for (const row of rows) {
  const company = String(row['Company'] || '').trim()
  if (!company) continue
  const slug = SLUG_OVERRIDES[company] ?? slugify(company)
  const tier = Number(row['Tier']) || 3
  const entry = {
    id: slug,
    companyName: company,
    application: String(row['Application'] || '').trim(),
    projectStatus: String(row['Project Status'] || '').trim(),
    tier,
    lastUpdated: null,
    notes: [],
  }
  if (tier === 1) tier1.push(entry)
  else if (tier === 2) tier2.push(entry)
  else tier3.push(entry)
}

mkdirSync(join(ROOT, 'public', 'data', 'rd'), { recursive: true })
const output = { generated: new Date().toISOString(), tier1, tier2, tier3 }
writeFileSync(OUT_PATH, JSON.stringify(output, null, 2))
console.log(`Written: ${OUT_PATH} (T1:${tier1.length} T2:${tier2.length} T3:${tier3.length})`)
```

### 1b — `scripts/sync-rd-files.mjs`

Scans each local `R&DDashboard/[folder]/` for non-README files, uploads them to DO Spaces at `rd-files/[slug]/[filename]`, and writes the file manifest to `public/data/rd/files.json`.

```javascript
// scripts/sync-rd-files.mjs
import { S3Client, PutObjectCommand, ListObjectsV2Command } from '@aws-sdk/client-s3'
import { readFileSync, readdirSync, writeFileSync, statSync } from 'fs'
import { join, dirname, extname, basename } from 'path'
import { fileURLToPath } from 'url'
import 'dotenv/config'

const __dirname = dirname(fileURLToPath(import.meta.url))
const ROOT = join(__dirname, '..')
const RD_ROOT = join(ROOT, '..', 'R&DDashboard')

// Folder → slug mapping
const FOLDER_SLUG_MAP = {
  'AB': 'ab',
  'Neptune': 'neptune',
  'Momentum': 'momentum',
  'Mayo-Clinic': 'mayo-clinic',
  'Fearsome': 'fearsome',
  'Aspero': 'aspero',
  'Alcon': 'alcon',
  'Abyrx': 'abyrx',
  'Boston-Scientific': 'boston-scientific',
  'Medicone': 'medicone',
  'P-Trap': 'p-trap',
  'Mazon': 'mazon',
  'Design-33': 'design-33',
  'Liqid-Medical': 'liqid-medical',
  'BMC': 'bmc',
}

const IGNORED_FILES = new Set(['README.md', '.DS_Store', 'Thumbs.db'])

const s3 = new S3Client({
  endpoint: `https://${process.env.DO_SPACES_REGION}.digitaloceanspaces.com`,
  region: process.env.DO_SPACES_REGION,
  credentials: {
    accessKeyId: process.env.DO_SPACES_KEY,
    secretAccessKey: process.env.DO_SPACES_SECRET,
  },
  forcePathStyle: false,
})

const manifest = {}

for (const [folder, slug] of Object.entries(FOLDER_SLUG_MAP)) {
  const folderPath = join(RD_ROOT, folder)
  let files
  try { files = readdirSync(folderPath) } catch { continue }

  manifest[slug] = []

  for (const file of files) {
    if (IGNORED_FILES.has(file) || file.startsWith('.')) continue
    const filePath = join(folderPath, file)
    if (!statSync(filePath).isFile()) continue

    const key = `rd-files/${slug}/${file}`
    const body = readFileSync(filePath)
    await s3.send(new PutObjectCommand({
      Bucket: process.env.DO_SPACES_BUCKET,
      Key: key,
      Body: body,
      ACL: 'private',
      ContentType: getContentType(extname(file)),
    }))
    manifest[slug].push({
      filename: file,
      description: '', // user can fill in or leave blank
      uploadedAt: new Date().toISOString(),
    })
    console.log(`  ✓ Uploaded: ${key}`)
  }
}

writeFileSync(join(ROOT, 'public', 'data', 'rd', 'files.json'), JSON.stringify(manifest, null, 2))
console.log('Written: public/data/rd/files.json')

function getContentType(ext) {
  const map = { '.pdf': 'application/pdf', '.xlsx': 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet', '.docx': 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', '.csv': 'text/csv', '.png': 'image/png', '.jpg': 'image/jpeg', '.zip': 'application/zip' }
  return map[ext.toLowerCase()] ?? 'application/octet-stream'
}
```

Identical pattern to NuSilq's `overlay.ts` but with a different localStorage key:

```typescript
import type { Overlay, RdProjectsData, NoteEntry, RdProject } from './types'

const OVERLAY_KEY = 'rd-notes-overlay'

export function loadOverlay(): Overlay { /* localStorage logic, same as NuSilq */ }
export function saveOverlay(overlay: Overlay): void { /* same as NuSilq */ }

export function mergeData(base: RdProjectsData, overlay: Overlay): RdProjectsData {
  function applyOverlay(items: RdProject[]): RdProject[] {
    return items.map(item => {
      const entry = overlay[item.id]
      if (!entry) return item
      return {
        ...item,
        ...(entry.baseOverride ?? {}),
        notes: [...item.notes, ...(entry.notes ?? [])],
        _edited: !!entry.baseOverride,
      }
    })
  }
  return {
    ...base,
    tier1: applyOverlay(base.tier1),
    tier2: applyOverlay(base.tier2),
    tier3: applyOverlay(base.tier3),
  }
}
```

### 4b — File Download API: `src/app/api/rd/download/route.ts`

Generates a presigned S3 URL so only users who have accessed the dashboard can trigger downloads (the URLs are short-lived and not guessable):

```typescript
import { NextResponse } from 'next/server'
import { S3Client, GetObjectCommand } from '@aws-sdk/client-s3'
import { getSignedUrl } from '@aws-sdk/s3-request-presigner'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

export async function GET(req: Request) {
  const { searchParams } = new URL(req.url)
  const slug = searchParams.get('slug')
  const file = searchParams.get('file')

  if (!slug || !file) {
    return NextResponse.json({ error: 'Missing slug or file' }, { status: 400 })
  }

  // Basic sanitization — prevent path traversal
  if (slug.includes('..') || file.includes('..') || file.includes('/')) {
    return NextResponse.json({ error: 'Invalid parameters' }, { status: 400 })
  }

  const s3 = new S3Client({
    endpoint: `https://${process.env.DO_SPACES_REGION}.digitaloceanspaces.com`,
    region: process.env.DO_SPACES_REGION!,
    credentials: {
      accessKeyId: process.env.DO_SPACES_KEY!,
      secretAccessKey: process.env.DO_SPACES_SECRET!,
    },
    forcePathStyle: false,
  })

  const key = `rd-files/${slug}/${file}`
  const cmd = new GetObjectCommand({
    Bucket: process.env.DO_SPACES_BUCKET!,
    Key: key,
    ResponseContentDisposition: `attachment; filename="${file}"`,
  })

  try {
    const url = await getSignedUrl(s3, cmd, { expiresIn: 3600 }) // 1 hour
    return NextResponse.redirect(url)
  } catch {
    return NextResponse.json({ error: 'File not found' }, { status: 404 })
  }
}
```
